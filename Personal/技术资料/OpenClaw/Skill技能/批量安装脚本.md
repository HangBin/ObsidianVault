  echo "Executing: $cmd"
  eval "$cmd"
  if [ $? -eq 0 ]; then
    echo "✅ Success: $cmd"
  else
    echo "❌ Failed: $cmd"
  fi
  echo "----------------------------------------"
done
EOF

赋予执行权限 
```chmod +x install_skills.sh```

运行脚本 
```./install_skills.sh```
