```bash
# 查看可用模型
openclaw models list
# 查看模型状态
openclaw models status
# 扫描可用模型
openclaw models scan

# 设置默认模型
# 腾讯云DeepSeek v3 0324
openclaw models set qcloudlkeap/deepseek-v3-0324
#openclaw models set minimax/MiniMax-M2.1 #未测试
#openclaw models set glm-4 #未测试
openclaw config set agents.defaults.model.primary "qcloudlkeap/deepseek-v3-0324"
 
# 启动交互式向导添加模型
openclaw onboard --auth-choice apiKey \ 
  --token-provider openrouter \ 
  --token "$OPENROUTER_API_KEY"

# 测试模型连接(有误)
openclaw models probe qcloudlkeap/deepseek-v3-0324
```
